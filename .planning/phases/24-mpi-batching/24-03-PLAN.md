# Plan 24-03: Worker-Side Batch Processing

## Objective

Modify worker_routines.cpp to receive and process batched tasks, executing all particles in a batch before sending a single completion signal.

## Context

Workers currently receive one Queue, process one particle, and send completion. This plan modifies them to:
1. Receive BatchedQueue with multiple particle IDs
2. Process all particles in the batch
3. Send single completion with count

Add processBatch function to handle batched tasks:

```cpp
// Phase 24: Process a batch of particles
void processBatch(BatchedQueue& batch) {
    Particle* ptcl;
    MPI_Request request;
    MPI_Status status;

    PROFILE_START(TimerID::WorkerTaskDispatch);

    switch (batch.task) {
        case TASK_IRR_FORCE:
            for (int i = 0; i < batch.count; i++) {
                ptcl = &particles[batch.pids[i]];
                ptcl->compute_acceleration_irr();
                ptcl->new_current_block_irr = ptcl->current_block_irr + ptcl->time_block_irr;
                ptcl->calculate_time_step_irr();
                ptcl->next_block_irr = ptcl->new_current_block_irr + ptcl->time_block_irr;
                ptcl->is_up_to_date = true;
            }
            break;

        case TASK_REG_FORCE:
            for (int i = 0; i < batch.count; i++) {
                ptcl = &particles[batch.pids[i]];
                ptcl->compute_acceleration_reg();
            }
            break;

        case TASK_IRR_UPDATE:
            for (int i = 0; i < batch.count; i++) {
                ptcl = &particles[batch.pids[i]];
                if (ptcl->num_neighbors != 0)
                    ptcl->update_particle();
                ptcl->current_block_irr = ptcl->new_current_block_irr;
                ptcl->current_time_irr = ptcl->current_block_irr * time_step;
            }
            break;

        case TASK_REG_UPDATE:
            for (int i = 0; i < batch.count; i++) {
                ptcl = &particles[batch.pids[i]];
                ptcl->current_block_reg += ptcl->time_block_reg;
                ptcl->current_time_reg = ptcl->current_block_reg * time_step;
                ptcl->update_particle();
                std::memcpy(neighbors + ptcl->neighbors_offset,
                           new_neighbors + ptcl->neighbors_offset,
                           sizeof(int) * ptcl->new_num_neighbors);
                ptcl->num_neighbors = ptcl->new_num_neighbors;
                ptcl->calculate_time_step_reg();
                ptcl->calculate_time_step_irr();
                ptcl->update_radius();
                ptcl->next_block_irr = ptcl->current_block_irr + ptcl->time_block_irr;
            }
            break;

        case TASK_INIT_ACC_1:
            for (int i = 0; i < batch.count; i++) {
                ptcl = &particles[batch.pids[i]];
                CalculateAcceleration01(ptcl);
            }
            break;

        case TASK_INIT_ACC_2:
            for (int i = 0; i < batch.count; i++) {
                ptcl = &particles[batch.pids[i]];
                CalculateAcceleration23(ptcl);
            }
            break;

        case TASK_INIT_TIME:
            for (int i = 0; i < batch.count; i++) {
                ptcl = &particles[batch.pids[i]];
                if (ptcl->is_active)
                    ptcl->initialize_time_step();
            }
            break;

        default:
            fprintf(stderr, "Unsupported batch task type: %d\n", static_cast<int>(batch.task));
            exit(EXIT_FAILURE);
    }

    PROFILE_STOP(TimerID::WorkerTaskDispatch);

    // Send single completion with batch count
    PROFILE_START(TimerID::WorkerSendComplete);
    int send_count = batch.count;
    MPI_Isend(&send_count, 1, MPI_INT, ROOT, TERMINATE_TAG, MPI_COMM_WORLD, &request);
    MPI_Wait(&request, &status);
    PROFILE_STOP(TimerID::WorkerSendComplete);
}
```
