```cpp
void Particle::compute_acceleration_irr() {
    // ... early exit check ...

    // Self prediction (already fast, skip timing)
    this->predict_particle_second_order(this->time_step_irr, pos, vel);

    int pairs_count = 0;  // Track neighbor pairs

    // Main neighbor loop
    PROFILE_START(TimerID::IrregularNeighborLoop);
    for (int i=0; i<this->num_neighbors; i++) {
        // ... existing loop code ...
        pairs_count++;  // Count each neighbor interaction
    }
    PROFILE_STOP(TimerID::IrregularNeighborLoop);
    PROFILE_WORK(TimerID::IrregularPairsEvaluated, pairs_count);

    // CM particle loop
    PROFILE_START(TimerID::IrregularCMLoop);
    for (int i: CMPtclsSet) {
        // ... existing CM loop code ...
        pairs_count++;
    }
    PROFILE_STOP(TimerID::IrregularCMLoop);
    PROFILE_WORK(TimerID::IrregularPairsEvaluated, CMPtclsSet.size());

    // 4th order correction
    PROFILE_START(TimerID::IrregularCorrection);
    for (int dim=0; dim<DIM; dim++) {
        // ... existing correction code ...
    }
    PROFILE_STOP(TimerID::IrregularCorrection);
}
```
