void Profiler::printLoadBalanceReport(std::ostream& os) const {
    os << "Timers with significant imbalance (max/avg > 1.2):\n";

    std::vector<std::pair<TimerID, double>> imbalanced;
    for (int i = 0; i < static_cast<int>(TimerID::NUM_TIMERS); ++i) {
        auto& agg = aggregated_stats_[i];
        if (agg.avg_seconds > 0.001) {  // Only significant timers
            double ratio = agg.max_seconds / agg.avg_seconds;
            if (ratio > 1.2) {
                imbalanced.emplace_back(static_cast<TimerID>(i), ratio);
            }
        }
    }

    // Sort by ratio descending
    std::sort(imbalanced.begin(), imbalanced.end(),
              [](auto& a, auto& b) { return a.second > b.second; });

    for (auto& [id, ratio] : imbalanced) {
        os << "  " << getTimerName(id) << ": "
           << std::fixed << std::setprecision(2) << ratio << "x"
           << " (slowest rank: " << aggregated_stats_[static_cast<int>(id)].max_rank << ")\n";
    }

    if (imbalanced.empty()) {
        os << "  No significant imbalance detected.\n";
    }
}
```

void Profiler::writeAggregatedJSON(const std::string& filename) const {
    std::ofstream file(filename);
    file << "{\n";
    file << "  \"num_ranks\": " << num_ranks_ << ",\n";
    file << "  \"aggregated_timers\": {\n";

    for (int i = 0; i < static_cast<int>(TimerID::NUM_TIMERS); ++i) {
        auto& agg = aggregated_stats_[i];
        file << "    \"" << getTimerName(static_cast<TimerID>(i)) << "\": {\n";
        file << "      \"min_seconds\": " << agg.min_seconds << ",\n";
        file << "      \"avg_seconds\": " << agg.avg_seconds << ",\n";
        file << "      \"max_seconds\": " << agg.max_seconds << ",\n";
        file << "      \"min_rank\": " << agg.min_rank << ",\n";
        file << "      \"max_rank\": " << agg.max_rank << "\n";
        file << "    }";
        if (i < static_cast<int>(TimerID::NUM_TIMERS) - 1) file << ",";
        file << "\n";
    }

    file << "  }\n";
    file << "}\n";
}
```
